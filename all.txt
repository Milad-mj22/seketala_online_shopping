import os
import sys

# پوشه‌هایی که به طور پیش‌فرض نادیده گرفته می‌شوند
SKIP_DIRS = {
    '__pycache__', '.git', '.venv', 'venv', 'env',
    'node_modules', '.idea', '.vscode', 'build', 'dist',
    '.mypy_cache', '.pytest_cache', '.tox', 'site-packages'
}


def ask_yes_no(question, default=False):
    """پرسش بله/خیر از کاربر"""
    suffix = " [y/N]: " if not default else " [Y/n]: "
    while True:
        answer = input(question + suffix).strip().lower()
        if answer == "":
            return default
        if answer in ("y", "yes", "بله", "ب", "آ", "اری"):
            return True
        if answer in ("n", "no", "خیر", "خ", "ن", "نه"):
            return False
        print("⚠️ لطفاً y یا n وارد کنید.")


def collect_python_files(root_dir, output_file):
    root_dir = os.path.abspath(root_dir)
    output_file = os.path.abspath(output_file)

    py_files = []
    skipped_dirs = []

    print("\n" + "=" * 60)
    print("🔍 پیمایش پوشه‌ها...")
    print("=" * 60)

    for dirpath, dirnames, filenames in os.walk(root_dir, topdown=True):
        rel = os.path.relpath(dirpath, root_dir)
        rel_display = "." if rel == "." else rel

        # بررسی پوشه‌های پیش‌فرض نادیده گرفته شده
        new_dirnames = []
        for d in dirnames:
            if d in SKIP_DIRS:
                skipped_dirs.append(os.path.join(rel_display, d))
                continue
            new_dirnames.append(d)
        dirnames[:] = new_dirnames

        # پرسش درباره‌ی خود پوشه (به جز ریشه)
        if dirpath != root_dir:
            if not ask_yes_no(f"📁 پوشه «{rel_display}» پردازش شود؟", default=True):
                skipped_dirs.append(rel_display)
                dirnames[:] = []  # زیرپوشه‌ها را پیمایش نکن
                continue

        # جمع‌آوری فایل‌های .py در این پوشه
        for fname in filenames:
            if fname.lower().endswith('.py'):
                py_files.append(os.path.join(dirpath, fname))

    py_files.sort()

    if not py_files:
        print(f"\n⚠️ هیچ فایل .py برای تجمیع پیدا نشد.")
        return

    total_bytes = 0

    with open(output_file, 'w', encoding='utf-8') as out:
        out.write("=" * 80 + "\n")
        out.write("📦 تجمیع فایل‌های پایتون\n")
        out.write(f"📁 دایرکتوری ریشه: {root_dir}\n")
        out.write(f"📄 تعداد فایل‌ها: {len(py_files)}\n")
        out.write("=" * 80 + "\n\n")

        for path in py_files:
            rel_path = os.path.relpath(path, root_dir)
            try:
                with open(path, 'r', encoding='utf-8') as f:
                    content = f.read()
            except UnicodeDecodeError:
                try:
                    with open(path, 'r', encoding='latin-1') as f:
                        content = f.read()
                except Exception as e:
                    out.write("\n" + "=" * 80 + "\n")
                    out.write(f"📄 فایل: {path}\n")
                    out.write(f"❌ خطا در خواندن فایل: {e}\n")
                    out.write("=" * 80 + "\n\n")
                    continue

            total_bytes += len(content.encode('utf-8'))

            out.write("\n" + "=" * 80 + "\n")
            out.write(f"📄 فایل: {path}\n")
            out.write(f"📁 مسیر نسبی: {rel_path}\n")
            out.write(f"📏 اندازه: {len(content)} کاراکتر\n")
            out.write("=" * 80 + "\n")
            out.write(content)
            if not content.endswith("\n"):
                out.write("\n")
            out.write("\n")

    print("\n" + "=" * 60)
    print(f"✅ {len(py_files)} فایل تجمیع شد.")
    print(f"📏 حجم کل: {total_bytes / 1024:.1f} KB")
    print(f"💾 فایل خروجی: {output_file}")

    if skipped_dirs:
        print(f"\n⏭️ پوشه‌های نادیده گرفته شده ({len(skipped_dirs)}):")
        for d in skipped_dirs:
            print(f"   - {d}")


def main():
    if len(sys.argv) >= 2:
        root_dir = sys.argv[1]
    else:
        root_dir = input("📁 مسیر دایرکتوری ریشه را وارد کنید: ").strip().strip('"').strip("'")

    if not os.path.isdir(root_dir):
        print(f"❌ دایرکتوری وجود ندارد: {root_dir}")
        sys.exit(1)

    if len(sys.argv) >= 3:
        output_file = sys.argv[2]
    else:
        default_out = os.path.join(os.path.dirname(os.path.abspath(root_dir)), "all_python_files.txt")
        output_file = input(f"💾 مسیر فایل خروجی (پیش‌فرض: {default_out}): ").strip().strip('"').strip("'")
        if not output_file:
            output_file = default_out

    collect_python_files(root_dir, output_file)


if __name__ == "__main__":
    main()

from django import forms
from .models import Complaint


class ComplaintForm(forms.ModelForm):
    class Meta:
        model = Complaint
        fields = ['name', 'phone', 'email', 'order_id', 'subject', 'message']

    def clean_phone(self):
        phone = self.cleaned_data.get('phone', '')
        digits = ''.join(c for c in phone if c.isdigit())
        if len(digits) < 10:
            raise forms.ValidationError('شماره تماس معتبر نیست.')
        return phone

    def clean_message(self):
        msg = self.cleaned_data.get('message', '').strip()
        if len(msg) < 5:
            raise forms.ValidationError('متن شکایت باید حداقل ۵ کاراکتر باشد.')
        return msg

from django.db import models



class Complaint(models.Model):
    STATUS_CHOICES = [
        ('new', 'جدید'),
        ('in_progress', 'در حال بررسی'),
        ('resolved', 'حل شده'),
        ('rejected', 'رد شده'),
    ]

    SUBJECT_CHOICES = [
        ('payment', 'مشکل در پرداخت'),
        ('delivery', 'مشکل در ارسال'),
        ('product', 'مشکل در کیفیت غذا'),
        ('support', 'مشکل با پشتیبانی'),
        ('other', 'سایر'),
    ]

    name        = models.CharField(max_length=100, verbose_name='نام')
    phone       = models.CharField(max_length=20, verbose_name='شماره تماس')
    email       = models.EmailField(blank=True, verbose_name='ایمیل')
    order_id    = models.CharField(max_length=100, blank=True, verbose_name='شماره سفارش')
    subject     = models.CharField(max_length=30, choices=SUBJECT_CHOICES, verbose_name='موضوع')
    message     = models.TextField(verbose_name='متن شکایت')
    status      = models.CharField(max_length=20, choices=STATUS_CHOICES, default='new', verbose_name='وضعیت')
    ip_address  = models.GenericIPAddressField(blank=True, null=True, verbose_name='IP')
    user_agent  = models.TextField(blank=True, verbose_name='User Agent')
    created_at  = models.DateTimeField(auto_now_add=True, verbose_name='تاریخ ثبت')

    class Meta:
        verbose_name = 'شکایت'
        verbose_name_plural = 'شکایات'
        ordering = ['-created_at']

    def __str__(self):
        return f'#{self.id} — {self.name} ({self.phone})'

@require_POST
def submit_complaint(request):
    """
    دریافت شکایت به صورت AJAX (JSON).
    فرانت‌اند باید با header 'X-CSRFToken' و body JSON بفرستد.
    """
    try:
        # پشتیبانی از هر دو حالت JSON و form-encoded
        if request.content_type == 'application/json':
            data = json.loads(request.body)
        else:
            data = request.POST.dict()

        form = ComplaintForm(data)
        if not form.is_valid():
            return JsonResponse({
                'success': False,
                'message': 'اطلاعات نامعتبر است.',
                'errors': form.errors,
            }, status=400)

        complaint = form.save(commit=False)

        # ذخیره‌ی IP و User Agent
        xff = request.META.get('HTTP_X_FORWARDED_FOR')
        complaint.ip_address = (
            xff.split(',')[0].strip() if xff else request.META.get('REMOTE_ADDR')
        )
        complaint.user_agent = request.META.get('HTTP_USER_AGENT', '')[:1000]
        complaint.status = 'new'
        complaint.save()

        return JsonResponse({
            'success': True,
            'message': 'شکایت شما با موفقیت ثبت شد. تا ۴۸ ساعت آینده با شما تماس می‌گیریم.',
            'complaint_id': complaint.id,
        }, status=201)

    except json.JSONDecodeError:
        return JsonResponse({
            'success': False,
            'message': 'داده‌های ارسالی معتبر نیست.',
        }, status=400)
    except Exception as e:
        return JsonResponse({
            'success': False,
            'message': f'خطا در ثبت شکایت: {str(e)}',
        }, status=500)

        وظیفه شما کمک به مشتریان برای ثبت سفارش، پاسخ به سوالات درباره منو،

        منوی رستوران شامل:
        - برگر طلایی: ۲۵۰,۰۰۰ تومان
        - پیتزا مخصوص: ۳۸۰,۰۰۰ تومان
        - ساندویچ مرغ: ۱۹۰,۰۰۰ تومان
        - سیب‌زمینی ویژه: ۱۲۰,۰۰۰ تومان
        - نوشابه: ۳۰,۰۰۰ تومان
        
        زمان تحویل: زیر ۲۰ دقیقه
        آدرس: تهران، خیابان ولیعصر، پلاک ۴۵
        تلفن: ۰۲۱-۱۲۳۴۵۶۷۸
        
        لطفاً به فارسی و با لحن دوستانه پاسخ بده.
        اگر کاربر سفارش داد، لیست سفارش‌ها رو جمع‌بندی کن و قیمت کل رو محاسبه کن.
        """

        response = openai.ChatCompletion.create(
            model="gpt-3.5-turbo",  # یا "gpt-4"
            messages=[
                {"role": "system", "content": system_prompt},
                {"role": "user", "content": user_text}
            ],
            temperature=0.7,
            max_tokens=500
        )

        reply_text = response.choices[0].message.content

        # ===== 2. تبدیل متن به صدا با OpenAI TTS =====
        try:
            # OpenAI TTS (نسخه جدید)
            from openai import OpenAI
            client = OpenAI(api_key=settings.OPENAI_API_KEY)
            
            speech_response = client.audio.speech.create(
                model="tts-1",
                voice="nova",  # گزینه‌های: alloy, echo, fable, onyx, nova, shimmer
                input=reply_text,
                speed=1.0
            )

            # ذخیره فایل صوتی
            audio_filename = f"voice_response_{hash(reply_text)}.mp3"
            audio_path = Path('media') / 'audio' / audio_filename
            audio_path.parent.mkdir(parents=True, exist_ok=True)
            
            # ذخیره فایل
            with open(audio_path, 'wb') as f:
                f.write(speech_response.content)
            
            audio_url = f"/media/audio/{audio_filename}"

        except Exception as tts_error:
            print(f"TTS Error: {tts_error}")
            audio_url = None

        # ===== 3. برگردوندن جواب =====
        return JsonResponse({
            'reply': reply_text,
            'audio_url': audio_url,
        })

    except json.JSONDecodeError:
        return JsonResponse({'error': 'JSON نامعتبر'}, status=400)
    except Exception as e:
        print(f"Error: {e}")
        return JsonResponse({'error': str(e)}, status=500)
